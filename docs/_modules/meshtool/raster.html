class Raster:
    """
    Raster object (regular 2D grid) with a spatial reference.
    For a bathymetry, by convention the internal z axis is oriented upwards.

    Build a raster with :meth:`from_file` or :meth:`from_arrays`.

    Attributes
    ----------
    data : (nn, 3) ndarray
        Node coordinates. ``z`` is filled with 0 if the input points are 2D. 
    x, y :

    ox, oy : (nn,) ndarray

    nx, ny : int

    dx, dy : (nc, 3) ndarray
    """

        self.filename       = None
        self.z_direction    = z_direction
        self._crs           = None

        self.read(file, bounds=bounds, crs_bounds=crs_bounds, pixel_margin=pixel_margin)

    @classmethod
    def from_arrays(cls, x, y, data, crs=None, z_direction="up"):
        raster = cls.__new__(cls)

        if z_direction not in ("up", "down"):
            raise ValueError("z_direction must be 'up' or 'down'.")

        raster.filename = None
        raster.z_direction = z_direction
        raster.crs = crs

        data = np.asarray(data)
        if z_direction == "down":
            data = -data              # internal convention: z up

        raster.set_raster(x, y, data)
        return raster



    def set_raster(self, x, y, data):
        """
        Sets the grid and resets the cache attributes.
        """
        self.reset_cache()

        self.x, self.y      = np.asarray(x), np.asarray(y)
        self.data           = np.asarray(data)
        
        if x.ndim != 1 or y.ndim != 1:
            raise ValueError("x and y must be 1D arrays.")

        if x.size < 2 or y.size < 2:
            raise ValueError("x and y must contain at least two points.")

        if data.shape != (y.size, x.size):
            raise ValueError(f"data must have shape {(y.size, x.size)}, got {data.shape}.")
    
        self.ox, self.oy    = self.x[0], self.y[0]
        self.nx, self.ny    = self.x.shape[0], self.y.shape[0]
        self.dx, self.dy    = (self.x[-1] - self.x[0]) / (self.nx - 1), (self.y[-1] - self.y[0]) / (self.ny - 1)

        self.print_info()


    # ------------------------------------------------------------------
    # Cache
    # ------------------------------------------------------------------

    def reset_cache(self):
        self._res               = None
        self._nodata_mask       = None
        self._gradient          = None
        self._flood_mask        = None
        self._flood_ssh         = None
        self._idx_closest       = None

    @property
    def res_x(self):
        if self._res is None:
            self._compute_resolution()
        return self._res[0]
 
    @property
    def res_y(self):
        if self._res is None:
            self._compute_resolution()
        return self._res[1]
 
    @property
    def res(self):
        return 0.5 * (self.res_x + self.res_y)

    @property
    def nodata_mask(self):
        if self._nodata_mask is None:
            self._nodata_mask = np.isnan(self.data)
        return self._nodata_mask

    @property
    def grad_x(self):
        if self._gradient is None:
            self._compute_gradient()
        return self._gradient[0]
 
    @property
    def grad_y(self):
        if self._gradient is None:
            self._compute_gradient()
        return self._gradient[1]
 
    @property
    def gradient(self):
        """Gradient norm. Not cached (only grad_x and grad_y are)."""
        return np.hypot(self.grad_x, self.grad_y)

    @property
    def _closest_indices(self):
        """(2, ny, nx) indices of the closest valid pixel of every pixel. Heavy: only built when needed."""
        if self._idx_closest is None:
            self._idx_closest = ndimage.distance_transform_edt(
                self.nodata_mask, return_distances=False, return_indices=True)
        return self._idx_closest

    @property
    def crs(self):
        """
        ``pyproj.CRS`` or None.
        """
        return self._crs
    
    @crs.setter
    def crs(self, value):
        self._crs = proj.to_crs(value)

    def _to_raster_crs(self, x, y, projection):
        """Return (x, y) float arrays expressed in the raster CRS."""
        
        if projection is None:
            return x, y

        if self.crs is None:
            raise ValueError("The raster has no CRS: cannot convert points from another projection. "
                             "Set `raster.crs` or pass projection=None.")

        return proj.transform_points(x, y, proj.to_crs(projection), self.crs)

    def read(self, file, bounds=None, crs_bounds=None, pixel_margin=10):
        """
        Read data from a Geotiff file, block by block, directly into a single float32 array.
        North-up files are flipped.
        If `bounds` is given, only the corresponding window is read.
        """
        cmn.comment(f"Reading raster file: {file}")
        cmn.comment(f"File vertical direction : {self.z_direction}")

        dataset = gdal.Open(str(file))

        file_crs = dataset.GetProjection()
        crs = proj.to_crs(file_crs) if file_crs else None

        gt = dataset.GetGeoTransform()

        if gt[2] != 0 or gt[4] != 0 or gt[1] <= 0:
            raise ValueError("Rotated or mirrored geotransforms are not supported.")
        
        file_nx = dataset.RasterXSize
        file_ny = dataset.RasterYSize
        dx = gt[1]
        dy = abs(gt[5])
        northup = gt[5] < 0

        # Center of the origin pixel
        file_ox = gt[0] + 0.5 * dx
        file_oy = gt[3] + (file_ny - 0.5) * gt[5] if northup else gt[3] + 0.5 * gt[5]

        # Window to read, if any
        if bounds is None:
            i0, i1, j0, j1 = 0, file_nx, 0, file_ny
        else:
            i0, i1, j0, j1 = _window_ids(bounds, crs_bounds, pixel_margin, file_ox, file_oy, dx, dy, file_nx, file_ny, crs)

        band = dataset.GetRasterBand(1)
        nodata = band.GetNoDataValue()

        data = band.ReadAsArray(i0, file_ny-j1 if northup else j0, i1-i0, j1-j0).astype(np.float32, copy=False)

        # Reorient data for northup convention
        if northup:
            data = np.ascontiguousarray(np.flipud(data))

        # Handle nodata values
        mask_nodata = ~np.isfinite(data)
        if nodata is not None:
             mask_nodata |= (data == nodata)
        data[mask_nodata] = np.nan              # Internal convention

        if self.z_direction == "down":
            data *= -1                          # Internal upwards z convention

        self.filename = str(file)
        self.crs = crs

        # data[0 , 0] : SW, data[ny, 0] : NW, data[0 ,nx] : SE, data[ny,nx] : NE
        self.set_raster(file_ox + dx * np.arange(i0, i1), file_oy + dy * np.arange(j0, j1), data)

        # Close the dataset
        dataset = None 

    def print_info(self):
        cmn.comment(f"{'Origin':<12} : Ox = {self.ox} , Oy = {self.oy}")
        cmn.comment(f"{'Pixel size':<12} : dx = {self.res_x:.2f} m, dy = {self.res_y:.2f} m")
        cmn.comment(f"{'Raster shape':<12} : {self.data.shape}")

    def _compute_resolution(self):
        """
        Pixel size in meters, evaluated at the center of the raster.
        """
        x0 = self.x[(self.nx - 1) // 2]
        y0 = self.y[(self.ny - 1) // 2]
 
        if self.crs is not None and self.crs.is_geographic:
            geod = Geod(ellps="WGS84")
            _, _, rx = geod.inv(x0, y0, x0 + self.dx, y0)
            _, _, ry = geod.inv(x0, y0, x0, y0 + self.dy)
        else:
            rx, ry = self.dx, self.dy
 
        self._res = (rx, ry)

    def crop_to_domain(self, bounds, crs_bounds=None, pixel_margin=10):
        """
        Crop raster to the extent of a rectangular domain + extra margin
        ``bounds`` = (xmin, xmax, ymin, ymax), expressed in ``crs_bounds`` (or in the raster CRS if None).
        """

        i0, i1, j0, j1 = _window_ids(bounds, crs_bounds, pixel_margin, self.ox, self.oy, self.dx, self.dy, self.nx, self.ny, self.crs)

        if i0 == 0 and i1 == self.nx and j0 == 0 and j1 == self.ny:
            return
        
        old_size = self.nx * self.ny

        self.set_raster(self.x[i0:i1], self.y[j0:j1], self.data[j0:j1, i0:i1].copy())

        new_size = self.nx * self.ny
        cmn.comment(f"\nRaster data cropped to the extent of the domain : {100*(1-new_size/old_size):.1f} % decrease in size")



    def _compute_gradient(self):
        """Raster field gradient (NaN values are propagated)."""
        dfy, dfx = np.gradient(self.data, self.res_y, self.res_x)  # The gradient propagates NaN value from the data array
        self._gradient = (dfx, dfy)

    def eval_slope(self, x, y, projection=None):
        x, y = self._to_raster_crs(x, y, projection)

        gx = numerics.interp_bilinear(x, y, self.x, self.y, self.grad_x)
        gy = numerics.interp_bilinear(x, y, self.x, self.y, self.grad_y)

        return np.sqrt(gx**2 + gy**2)

    def smooth(self, size_smooth):
        stdev = size_smooth/self.res
        self.data = numerics.unif_smooth(data=self.data, stdev=stdev)
        self.reset_cache()

    def compute_flood_fill(self, ssh_target=0.):
        """
        Determine the parts of the terrain that are covered in water, for a given ssh level.
        Flood fill starting from the lowest bathymetry point.
        """
        seed_index = np.nanargmin(self.data)
        i, j = np.unravel_index(seed_index, self.data.shape)

        seed_mask = np.zeros_like(self.data, dtype=bool)
        seed_mask[i,j] = True

        self._flood_mask = ndimage.binary_propagation(seed_mask, mask=(self.data < ssh_target))
        self._flood_ssh = ssh_target

    def is_flooded(self, x, y, projection=None, ssh_target=None):
        if ssh_target is None:
            ssh_target = 0. if self._flood_ssh is None else self._flood_ssh

        if self._flood_mask is None or self._flood_ssh != ssh_target:
            self.compute_flood_fill(ssh_target)
        
        x, y = self._to_raster_crs(x, y, projection)

        im = np.clip( ((x - self.ox)/self.dx).astype(np.int32), 0, self.nx-2)
        jm = np.clip( ((y - self.oy)/self.dy).astype(np.int32), 0, self.ny-2)
        ip = im+1
        jp = jm+1

        fm = self._flood_mask
        return (fm[jm, im] | fm[jp, im] | fm[jm, ip] | fm[jp, ip])

    def eval(self, x, y, projection=None, fill_value=np.nan, fill_type='closest'):
        """
        Bilinear interpolation of the raster at the points (x, y).

        fill_type = 'value' or 'closest'
        """  
        if fill_type not in ("value", "closest"):
            raise ValueError("Incorrect raster evaluation argument 'fill_type'")
        
        x, y = self._to_raster_crs(x, y, projection)

        if fill_type == 'value':
            return numerics.interp_bilinear(x, y, self.x, self.y, self.data, fill_value=fill_value)

        values, mask_valid = numerics.interp_bilinear(x, y, self.x, self.y, self.data, fill_value=fill_value, return_mask=True)
        missing = ~mask_valid

        if missing.any():
            cmn.comment(f"WARNING : {missing.sum()} cells had no corresponding raster value and were replaced with the closest one.")
            values[missing] = self._eval_closest(x[missing], y[missing])

        return values



    def _eval_closest(self, x, y):
        """Value of the closest valid pixel of each point."""
        i = np.clip( np.round((x - self.ox)/self.dx).astype(int), 0, self.nx-1)
        j = np.clip( np.round((y - self.oy)/self.dy).astype(int), 0, self.ny-1)

        idx = self._closest_indices

        return self.data[idx[0, j, i], idx[1, j, i]]

    def __call__(self, x, y, projection=None):
        return self.eval(x, y, projection)




def _window_ids(bounds, crs_bounds, pixel_margin, ox, oy, dx, dy, nx, ny, crs,):
    if crs_bounds is not None:
        if crs is None:
            raise ValueError("The raster has no CRS: cannot interpret bounds given in another CRS.")
        xmin, xmax, ymin, ymax = proj.transform_bounds(bounds, crs_bounds, crs)
    else:
        xmin, xmax, ymin, ymax = bounds

    i0 = max(int(np.floor((xmin - ox) / dx)) - pixel_margin, 0)
    i1 = min(int(np.ceil((xmax - ox) / dx)) + pixel_margin + 1, nx)
    j0 = max(int(np.floor((ymin - oy) / dy)) - pixel_margin, 0)
    j1 = min(int(np.ceil((ymax - oy) / dy)) + pixel_margin + 1, ny)

    if i1 <= i0 or j1 <= j0:
        raise ValueError("Problem in the input bounds.")

    return i0, i1, j0, j1

def read_regional_depth(fname, z_direction="up"):
    """
    Read bathymetry in a regional.depth(-ele/-node).a binary file.
    By convention, the internal z axis is oriented upwards.
    """
    if z_direction not in ("up", "down"):
        raise ValueError("z_direction must be 'up' or 'down'.")

    fname = Path(fname)

    if fname.suffix != ".a":
        raise ValueError("Regional file containing bathymetry must end with '.a'")

    cmn.comment(f"Reading regional depth bathymetry file: {fname}.")
    bathy = np.fromfile(fname, dtype=">f4").astype(np.float32)

    if z_direction == "down":
        bathy *= -1

    return bathy
