class Mesh:
    """
    Mesh representation based on the `meshio.Mesh` object.
    Quad elements are not supported.

    Build a mesh with :meth:`from_file` or :meth:`from_arrays`.

    Attributes
    ----------
    node_coords : (nn, 3) ndarray
        Node coordinates. ``z`` is filled with 0 if the input points are 2D.    
    x, y, z : (nn,) ndarray
        Views of the columns of ``node_coords``.
    nn, nc, ne : int
        Number of nodes, triangle and line cells.
    connectivity : (nc, 3) ndarray
        0-based node indices of each triangle.
    edge_nodes : (ne, 2) ndarray
        0-based node indices of each line cell (empty if there is none).
    physical_names : dict[str, list[tuple[int, int]]]
        Mapping from physical group name to the list of its `(dim, tag)` pairs, as read
        from the `$PhysicalNames` section of the file. A name shared by several physical
        groups maps to several pairs. Empty if the section is absent.
    metadata : dict[str, str]
        Custom Gmsh blocks, preserved when writing a ``.msh`` file.
    meshio_mesh : meshio.Mesh
        Underlying mesh object.

    Notes
    -----
    **Geometry depends on the CRS.** With a geographic CRS (lon/lat in degrees),
    areas and perimeters are computed on the sphere from 3D coordinates, while point
    location and interpolation are done in the (lon, lat) plane. With a projected CRS
    or ``crs=None``, everything is planar, in the units of the coordinates: areas are
    then true m² only if the projection preserves distances.

    The antimeridian is not handled: a mesh crossing it is not supported.
    """

    # ------------------------------------------------------------------
    # Construction
    # ------------------------------------------------------------------

    def __init__(self, file=None, meshio_mesh=None, crs=None):
        if file is not None and meshio_mesh is not None:
            raise ValueError("Specify either file or meshio_mesh, not both.")
        if file is None and meshio_mesh is None:
            raise ValueError("Either file or meshio_mesh must be provided.")

        self.mesh_file = file
        self.metadata = {}
        self.physical_names = {}
        self.version = self.binary = None
        self._crs = None

        if meshio_mesh is not None:
            self._set_mesh(meshio_mesh)
            self.crs = crs

        else:
            self._read(file, crs=crs)

    @classmethod
    def from_arrays(cls, node_coords, connectivity, crs=None):
        """
        Build a triangle mesh from arrays.

        Parameters
        ----------
        node_coords : array_like, shape (nn, 2) or (nn, 3)
            Node coordinates.
        connectivity : array_like of int, shape (nc, 3)
            0-based node indices of each triangle.
        crs : pyproj.CRS, EPSG code, WKT string or None

        Returns
        -------
        Mesh
        """
        node_coords = np.array(node_coords, dtype=np.float64)
        connectivity = np.array(connectivity, dtype=np.int32)

        if connectivity.ndim != 2 or connectivity.shape[1] != 3:
            raise ValueError("connectivity must have shape (nc, 3).")
        
        mesh = meshio.Mesh(points=node_coords, cells={"triangle": connectivity})
        return cls(meshio_mesh=mesh, crs=crs)



    def _set_mesh(self, meshio_mesh: meshio.Mesh):
        self.meshio_mesh = meshio_mesh

        # Nodes (fill z if the points are 2D)
        pts = np.asarray(meshio_mesh.points)
        if pts.shape[1] == 2:
            pts = np.column_stack((pts, np.zeros(len(pts))))
            meshio_mesh.points = pts

        self.node_coords = pts
        self.x = self.node_coords[:, 0]
        self.y = self.node_coords[:, 1]
        self.z = self.node_coords[:, 2]
        self.nn = len(self.node_coords)

        # Elements
        self.connectivity = self._get_meshio_cells("triangle")
        self.nc = len(self.connectivity)

        self.edge_nodes = self._get_meshio_cells("line")
        self.ne = len(self.edge_nodes)

        self.reset_cache()

    def _get_meshio_cells(self, cell_type):
        blocks = [b.data for b in self.meshio_mesh.cells if b.type == cell_type]
        if not blocks:
            return np.empty((0, 2 if cell_type == "line" else 3), dtype=np.int32)
        if len(blocks) != 1:
            raise ValueError(f"Mesh must contain a single '{cell_type}' cell block.")
        return blocks[0]

    @property
    def cell_neighbors(self):
        """
        (3, nc) int ndarray: Neighbor cells. ``cell_neighbors[k, i]`` is the cell sharing the 
        edge of cell ``i`` opposite to its local node ``k``, or -1 on the boundary.        
        """
        if self._cell_neighbors is None:
            self._compute_cell_neighbors()
        return self._cell_neighbors

    @property
    def boundary_edges(self):
        """
        Edges belonging to a single triangle (mesh boundary), and their triangle.

        Returns
        -------
        edge_nodes : (nb, 2) int ndarray
            0-based node indices of each boundary edge. The orientation of the triangle is kept.
        cell_ids : (nb,) int ndarray
            Triangle owning each edge.
        """
        boundary = (self.cell_neighbors == -1)
        local_ids, cell_ids = np.nonzero(boundary)
        edge_nodes = self.connectivity[cell_ids[:, None], (local_ids[:, None] + [1, 2]) % 3]
        return edge_nodes, cell_ids

    @property
    def node_weight(self):
        """Sum of the areas of the cells surrounding each node (0 for isolated nodes)."""
        if self._node_weight is None:
            w = np.zeros(self.nn)
            for k in range(3):
                w += np.bincount(self.connectivity[:, k], weights=self.cell_areas, minlength=self.nn)
            self._node_weight = w
        return self._node_weight

    @property
    def cell_to_node_operator(self):
        """
        Sparse (nn x nc) matrix W such that `W @ data_cells` is the area-weighted
        average of the surrounding cells at each node. Rows of isolated nodes are empty.
        """
        if self._cell_to_node_op is None:
            w = self.node_weight
            inv = np.zeros(self.nn)
            np.divide(1.0, w, out=inv, where=w > 0)

            rows = self.connectivity.ravel()                 # node of each (cell, local node) pair
            cols = np.repeat(np.arange(self.nc), 3)          # matching cell index
            vals = np.repeat(self.cell_areas, 3) * inv[rows]

            self._cell_to_node_op = sparse.csr_matrix((vals, (rows, cols)), shape=(self.nn, self.nc))
        return self._cell_to_node_op

    def _to_mesh_crs(self, x, y, points_crs):
        """Return (x, y) float arrays expressed in the mesh CRS."""
        if points_crs is None:
            return x, y
 
        if self.crs is None:
            raise ValueError("The mesh has no CRS: cannot convert points from another projection. "
                             "Set `mesh.crs` or pass points_crs=None.")
 
        return proj.transform_points(x, y, points_crs, self.crs)

        """

    def reproject(self, new_crs):
        """
        Transform the node coordinates to ``new_crs``, **in place**, and update ``crs``.

        Caches are reset. ``z`` is unchanged.
        """
        if self.crs is None:
            raise ValueError("Mesh has no CRS; declare one first with `mesh.crs = ...`.")
 
        x, y = proj.transform_points(self.x, self.y, self.crs, proj.to_crs(new_crs))
         
        self.meshio_mesh.points[:, 0] = x
        self.meshio_mesh.points[:, 1] = y

        self._set_mesh(self.meshio_mesh)  # refreshes x/y views and resets all caches
        self.crs = new_crs

    def get_physical_group(self, name, cell_type):
        """
        Cells of a physical group.

        Parameters
        ----------
        name : str
            Physical group name. A name shared by several tags is fully handled.
        cell_type : {"triangle", "line", ...}
            Cell type to extract.

        Returns
        -------
        (n, k) int ndarray
            Node indices of the selected cells (k = 3 for triangles, 2 for lines).
        """
        if name not in self.physical_names:
            raise ValueError(f"Physical group '{name}' not found. Available groups: {self.physical_names}")

        physical_tags = [tag for _, tag in self.physical_names[name]]

        selected = []
        for cells, physical_data in zip(self.meshio_mesh.cells, self.meshio_mesh.cell_data["gmsh:physical"]):
            if cells.type == cell_type:
                selected.append(cells.data[np.isin(physical_data, physical_tags)])

        if not selected:
            raise ValueError(f"Physical group '{name}' contains no '{cell_type}' cells.")

        return np.concatenate(selected, axis=0)



    def remove_1d_physical_group(self, names):
        """
        Remove 1D elements associated with the given physical groups.

        All 1D (dim, tag) pairs registered under each name in ``physical_names`` are removed,
        so names shared by several groups are fully handled. Non-1D groups sharing the same
        name are kept. ``physical_names`` and ``field_data`` are updated accordingly.

        Parameters
        ----------
        names : str or iterable of str
            Names of the 1D physical groups to remove.
        """
        names = [names] if isinstance(names, str) else list(names)

        # Collect every 1D tag behind these names (use self.physical_names because it keeps duplicates, 
        # whereas field_data does not)
        tag_ids = []
        for name in names:
            if name not in self.physical_names:
                raise ValueError(f"Physical group '{name}' not found. Available groups: {list(self.physical_names)}")
            tags = [tag for dim, tag in self.physical_names[name] if dim == 1]
            if not tags:
                raise ValueError(f"Physical group '{name}' has no 1D definition.")
            tag_ids.extend(tags)

        mesh = self.meshio_mesh
        for i, cell_block in enumerate(mesh.cells):
            if cell_block.type != "line":
                continue

            physical = mesh.cell_data["gmsh:physical"][i]
            keep = ~np.isin(physical, tag_ids)

            mesh.cells[i] = meshio.CellBlock("line", cell_block.data[keep])

            for key in mesh.cell_data:
                mesh.cell_data[key][i] = mesh.cell_data[key][i][keep]

        # Update the groups
        for name in names:
            remaining = [(dim, tag) for dim, tag in self.physical_names.get(name, []) if dim != 1]
            if remaining:
                self.physical_names[name] = remaining
                dim, tag = remaining[0]
                mesh.field_data[name] = np.array([tag, dim])
            else:
                self.physical_names.pop(name, None)
                mesh.field_data.pop(name, None)

        self.edge_nodes = self._get_meshio_cells("line")
        self.ne = len(self.edge_nodes)

        cmn.comment(f"Removed 1D physical groups {names} (tags {tag_ids}).")

    def write_ww3(self, filename, node_depth, ob_names):
        """
        Write a WaveWatch III compatible mesh.
        Selected boundary line cells are converted to vertex cells.
        Inspired from matlab code `WW3_mesh_generator.m`
        """
        if np.any(self.node_weight <= 0):
            raise ValueError("Isolated nodes are present in the mesh. Clean the mesh before converting to wavewatch format")
        
        ob_names = [ob_names] if isinstance(ob_names, str) else list(ob_names)

        ob = []
        for n in ob_names:
            ob.append(self.get_physical_group(n, "line"))
        ob_nodes = np.concatenate(ob).ravel()
        ob_nodes = np.unique(ob_nodes)

        lon, lat = self.get_lon_lat()

        mesh_io.write_ww3(filename, lon, lat, node_depth, self.connectivity, ob_nodes)

    def _compute_cell_surf_peri(self):
        """
        Compute triangle surface areas and perimeters.
        """
        nodes = self.connectivity

        if self.crs is not None and self.crs.is_geographic:
            xyz = np.column_stack(proj.lonlat_to_xyz(self.x, self.y))
            p1 = xyz[nodes[:, 0]]
            p2 = xyz[nodes[:, 1]]
            p3 = xyz[nodes[:, 2]]            
            self._cell_areas = 0.5 * np.linalg.norm(np.cross(p2-p1, p3-p1), axis=1)
        
        else:
            xy = np.column_stack((self.x, self.y))
            p1 = xy[nodes[:, 0]]
            p2 = xy[nodes[:, 1]]
            p3 = xy[nodes[:, 2]]
            v1 = p2-p1
            v2 = p3-p1
            self._cell_areas = 0.5 * np.abs(v1[:, 0] * v2[:, 1] - v1[:, 1] * v2[:, 0])

        self._cell_perimeters = np.linalg.norm(p2 - p1, axis=1) + np.linalg.norm(p3 - p2, axis=1) + np.linalg.norm(p1 - p3, axis=1)

    def barycentric_coordinates(self, x, y, cell_ids):
        """
        Barycentric coordinates (alpha, beta, gamma) of the points (x, y) in the given cells,
        i.e. the weights of the cell's local nodes (0, 1, 2) in this order.
        Degenerate cells give NaN.
        """
        nodes = self.connectivity[cell_ids]

        ax = self.x[nodes[:, 0]]
        ay = self.y[nodes[:, 0]]
        bx = self.x[nodes[:, 1]]
        by = self.y[nodes[:, 1]]
        cx = self.x[nodes[:, 2]]
        cy = self.y[nodes[:, 2]]

        v0x = bx - ax
        v0y = by - ay
        v1x = cx - ax
        v1y = cy - ay
        v2x = x - ax
        v2y = y - ay
        den = v0x * v1y - v1x * v0y

        beta = (v2x * v1y - v1x * v2y) / den
        gamma = (v0x * v2y - v2x * v0y) / den
        alpha = 1.0 - beta - gamma

        return alpha, beta, gamma

    def _compute_cell_neighbors(self):
        """
        cell_neighbors[k, i] = neighbor of cell 'i' through the edge opposed to the local node of index 'k' (-1 on the boundary).
        """
        n = self.nc
        nv = self.connectivity.max() + 1

        keys = np.empty(3 * n, dtype=np.int64)
        for k in range(3):
            a = self.connectivity[:, (k + 1) % 3]
            b = self.connectivity[:, (k + 2) % 3]
            s = keys[k * n:(k + 1) * n]
            np.minimum(a, b, out=s)
            s *= nv
            s += np.maximum(a, b)

        order = np.argsort(keys)
        keys = keys[order]
        pos = np.flatnonzero(keys[1:] == keys[:-1])
        if pos.size > 1 and np.any(pos[1:] - pos[:-1] == 1):
            raise ValueError("One edge has more than 2 neighboring cells")

        p, q = order[pos], order[pos + 1]
        self._cell_neighbors = np.full(3 * n, -1, dtype=np.int32)
        self._cell_neighbors[p] = q % n
        self._cell_neighbors[q] = p % n
        self._cell_neighbors = self._cell_neighbors.reshape(3, n)

    def add_cell_data(self, name, values, cell_type="triangle"):
        """Attach a per-cell field to the mesh, stored in ``meshio_mesh.cell_data[name]``."""
        values = np.asarray(values)
        n_target = sum(block.data.shape[0] for block in self.meshio_mesh.cells if block.type == cell_type)
        if n_target == 0:
            raise ValueError(f"The mesh contains no '{cell_type}' cells.")
        if values.shape[0] != n_target:
            raise ValueError(f"values has {values.shape[0]} elements, but the mesh has {n_target} '{cell_type}' cells.")

        field = []
        offset = 0
        for block in self.meshio_mesh.cells:
            n = block.data.shape[0]
            if block.type == cell_type:
                field.append(values[offset:offset + n])
                offset += n
            else:
                field.append(np.full(n, np.nan))

        self.meshio_mesh.cell_data[name] = field

    def cell_to_node(self, data_cells):
        """Area-weighted average of cell data onto nodes."""
        data = np.asarray(data_cells, dtype=np.float64)
        if data.shape[0] != self.nc:
            raise ValueError(f"data_cells has {data.shape[0]} rows, expected {self.nc} (one per cell).")
        out = self.cell_to_node_operator @ data
        out[self.node_weight <= 0.] = np.nan
        return out

    def node_to_cell(self, data_nodes):
        """Average of the node values over the 3 nodes of each cell."""
        data = np.asarray(data_nodes)
        if data.shape[0] != self.nn:
            raise ValueError(f"data_nodes has {data.shape[0]} rows, expected {self.nn} (one per node).")
        c = self.connectivity
        return (data[c[:, 0]] + data[c[:, 1]] + data[c[:, 2]]) / 3.0

    def locate(self, x, y, points_crs=None):
        """
        Find the triangle containing each point.

        Parameters
        ----------
        x, y : array_like
            Point coordinates, expressed in ``points_crs`` (or in the mesh CRS if None).
        points_crs : Any format

        Returns
        -------
        cell_ids : ndarray
            Cell index containing each point, or -1 if outside the mesh.
        """
        x = np.asarray(x)
        y = np.asarray(y)
        if x.shape != y.shape:
            raise ValueError("x and y must have the same shape.")
        
        x, y = self._to_mesh_crs(x, y, proj.to_crs(points_crs))
        return self.cell_locator.locate(x, y)

    def interp_node_data(self, x, y, data_nodes, points_crs=None, fill_value = np.nan):
        """
        Interpolate node data on the given (x,y) points using barycentric interpolation.
        Points outside the mesh are assigned ``fill_value``.
        """
        x = np.asarray(x)
        y = np.asarray(y)
        data_nodes = np.asarray(data_nodes)
        if x.shape != y.shape:
            raise ValueError("x and y must have the same shape.")
        
        if len(data_nodes) != self.nn:
            raise ValueError(f"data_nodes must have shape ({self.nn},), got {data_nodes.shape}.")

        # Project (x, y) to the mesh crs
        x, y = self._to_mesh_crs(x, y, proj.to_crs(points_crs))
        cell_ids = self.locate(x, y)

        inside = (cell_ids >= 0)
        cid = cell_ids[inside]
        alpha, beta, gamma = self.barycentric_coordinates(x[inside], y[inside], cid)
        n0, n1, n2 = self.connectivity[cid].T

        interp_data = np.full_like(x, fill_value, dtype=np.float64)
        interp_data[inside] = alpha * data_nodes[n0] + beta * data_nodes[n1] + gamma * data_nodes[n2]
        return interp_data

    def _filter_cells(self, keep):
        """
        Rebuild the mesh keeping only the triangles where ``keep`` is True.
        Nodes left without a kept triangle are dropped, as are the line cells that touch them.
        Point and cell data are sliced accordingly.
        """
        keep = np.asarray(keep, dtype=bool)
        if keep.shape != (self.nc,):
            raise ValueError(f"'keep' must have shape ({self.nc},).")

        m = self.meshio_mesh

        used = np.zeros(self.nn, dtype=bool)
        used[self.connectivity[keep]] = True
        new_id = np.cumsum(used) - 1                      # old node index -> new node index

        cells = []
        cell_data = {name: [] for name in m.cell_data}
        for i, block in enumerate(m.cells):
            mask = keep if block.type == "triangle" else used[block.data].all(axis=1)
            cells.append((block.type, new_id[block.data[mask]].astype(block.data.dtype)))
            for name, arrays in m.cell_data.items():
                cell_data[name].append(arrays[i][mask])

        self._set_mesh(meshio.Mesh(points=m.points[used],
                                cells=cells,
                                point_data={k: v[used] for k, v in m.point_data.items()},
                                cell_data=cell_data,
                                field_data=m.field_data))

    def remove_cells(self, cell_ids):
        """Remove triangles by index. Nodes that no longer belong to any triangle are removed too."""
        keep = np.ones(self.nc, dtype=bool)
        keep[np.atleast_1d(cell_ids)] = False
        self._filter_cells(keep)

    def remove_nodes(self, node_ids):
        """Remove nodes by index, together with all the triangles that use them."""
        self._filter_cells(~np.isin(self.connectivity, node_ids).any(axis=1))

    def remove_orphan_nodes(self):
        """Remove the nodes that belong to no triangle."""
        self._filter_cells(np.ones(self.nc, dtype=bool))

    def merge_nodes(self, drop, keep):
        """
        Replace node ``drop`` by node ``keep`` everywhere (typically a duplicated node).
        Triangles that become degenerate are removed.
        """
        for block in self.meshio_mesh.cells:
            block.data[block.data == drop] = keep
        c = self.connectivity
        degenerate = (c[:, 0] == c[:, 1]) | (c[:, 1] == c[:, 2]) | (c[:, 2] == c[:, 0])
        self._filter_cells(~degenerate)


    
class _CellLocator:
    """
    Spatial index for locating points in a triangular mesh.

    This class is an internal implementation detail of ``Mesh``.
    """

    CHUNK_SIZE = 500_000

    def __init__(self, mesh: Mesh):
        self.mesh = mesh

        # Build the spatial index from the mesh geometry (the bounding boxes of the cells).
        conn = mesh.connectivity
        x = mesh.x
        y = mesh.y

        xmin = np.minimum.reduce( [x[conn[:, 0]], x[conn[:, 1]], x[conn[:, 2]]] )
        xmax = np.maximum.reduce( [x[conn[:, 0]], x[conn[:, 1]], x[conn[:, 2]]] )
        ymin = np.minimum.reduce( [y[conn[:, 0]], y[conn[:, 1]], y[conn[:, 2]]] )
        ymax = np.maximum.reduce( [y[conn[:, 0]], y[conn[:, 1]], y[conn[:, 2]]] )

        P = box(xmin, ymin, xmax, ymax)
        
        self.tree = STRtree(P)

    def _find_cells(self, x, y):
        N = len(x)
        cell_ids = np.full(N, -1, dtype=np.int32)

        for start in range(0, N, self.CHUNK_SIZE):
            end = min(start + self.CHUNK_SIZE, N)

            x_chunk = x[start:end]
            y_chunk = y[start:end]

            query_points = points(x_chunk, y_chunk)
        
            point_ids, tri_ids = self.tree.query(query_points)

            px = x_chunk[point_ids]
            py = y_chunk[point_ids]

            alpha, beta, gamma = self.mesh.barycentric_coordinates(px, py, tri_ids)
            inside = (alpha >= 0) & (beta >= 0) & (gamma >= 0)
            
            cell_ids[start + point_ids[inside]] = tri_ids[inside]

        return cell_ids

    def locate(self, x, y):
        """
        Locate points in the mesh (already in the mesh CRS).

        Parameters
        ----------
        x, y : array_like
            Coordinates of the points.

        Returns
        -------
        cell_ids : ndarray
            Index of the containing triangle, or -1 if outside.
        """
        x = np.asarray(x)
        y = np.asarray(y)

        if x.shape != y.shape:
            raise ValueError("x and y must have the same shape.")

        cell_ids = self._find_cells(x.ravel(), y.ravel())

        return cell_ids.reshape(x.shape)
